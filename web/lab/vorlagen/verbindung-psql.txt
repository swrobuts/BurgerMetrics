psql "postgresql://studi_daba:thws@supabase.butscher.cloud:5433/postgres"

# Danach, in der psql-Sitzung:

\dn                     -- alle Schemata auflisten (relevant: wawi, burgermetrics)
\dt wawi.*              -- Tabellen im Schema wawi auflisten
\dm burgermetrics.*     -- materialisierte Sichten im Schema burgermetrics auflisten
\q                      -- Sitzung beenden
