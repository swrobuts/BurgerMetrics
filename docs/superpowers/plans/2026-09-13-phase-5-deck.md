```python
  1 Einleitungen einsammeln: Shapes im Einleitungsband (top 82–115 pt, width > 700 pt),
    Text normalisiert, alles unter 40 Zeichen verworfen — die Liste wird vollständig
    ausgegeben, damit jeder Eintrag gelesen wird.
  2 Zwei Mustersuchen über ALLE Texte (Floskeln, Meta-Sätze), je Treffer 70 Zeichen davor
    und 90 danach.
  3 Wortgleiche Einleitungen.

Ändert nichts. Aufruf: python3 pruefe_folientext.py deck.pptx
Rückgabe 1, wenn ein Treffer aus 2 oder 3 vorliegt.
"""
import re
import sys
from pptx import Presentation
from pptx.util import Pt

FLOSKELN = [
    r"spielt eine (wichtige|zentrale|entscheidende|große) Rolle", r"gewinn\w* (zunehmend )?an Bedeutung",
    r"nicht mehr wegzudenken", r"in der heutigen Zeit|heutzutage|mehr denn je", r"immer wichtiger",
    r"Dreh- und Angelpunkt|das A und O|Gamechanger|Quantensprung", r"Es ist wichtig|Grundsätzlich ist|Es gilt,",
    r"ganzheitlich|nachhaltig\w*|vielfältig\w*", r"revolutionier\w+|rasant",
    r"Die (Grafik|Abbildung|Tabelle|Folie) zeigt", r"Mehrwert\w*", r"Erfolgsfaktor\w*", r"Herausforderung\w*",
]
META = [
    r"dieses Kapitels?|vorige[nr]? Kapitel", r"die (bisherigen|gezeigten|obigen) ", r"wie (bereits|eingangs|oben) (erwähnt|gesehen)",
    r"Im Folgenden|Abschließend|Zusammenfassend lässt sich", r"wirkt wie ein|gleicht einem|ist wie ein",
    r"Der Stoff|in diesem Kurs|diese Folie", r"setzen den Maßstab", r"nicht \w+, sondern",
]


def texte(prs):
    # Liefert (Foliennummer, Shape-Name, top in pt, width in pt, Text) für jedes Shape mit Text
    for nr, folie in enumerate(prs.slides, start=1):
        for sh in folie.shapes:
            if not sh.has_text_frame:
                continue
            t = re.sub(r"\s+", " ", sh.text_frame.text).strip()
            if t:
                top = (sh.top or 0) / 12700
                width = (sh.width or 0) / 12700
                yield nr, sh.name, top, width, t


def main(pfad):
    prs = Presentation(pfad)
    alle = list(texte(prs))
    befund = 0

    print("== 1 Einleitungen (top 82–115, width > 700, ab 40 Zeichen)")
    einleitungen = [(nr, name, t) for nr, name, top, width, t in alle
                    if 82 <= top <= 115 and width > 700 and len(t) >= 40]
    for nr, name, t in einleitungen:
        print(f"Folie {nr:2d} [{name}] ({len(t)} Zeichen): {t}")

    print("\n== 2 Mustersuche: Floskeln und Meta-Sätze")
    for nr, name, top, width, t in alle:
        for muster in FLOSKELN + META:
            for m in re.finditer(muster, t, flags=re.IGNORECASE):
                a, e = max(0, m.start() - 70), min(len(t), m.end() + 90)
                print(f"Folie {nr:2d} [{name}] /{muster}/: …{t[a:e]}…")
                befund += 1

    print("\n== 3 Wortgleiche Einleitungen")
    gesehen = {}
    for nr, name, t in einleitungen:
        if t in gesehen:
            print(f"Folie {nr} wiederholt die Einleitung von Folie {gesehen[t]}")
            befund += 1
        gesehen.setdefault(t, nr)

    print(f"\n{len(einleitungen)} Einleitungen, {befund} Befund(e) aus Mustersuche und Wiederholung.")
    return 1 if befund else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1]))
```
